hostname R1
no ip domain-lookup
ip domain-name lab.local
username admin privilege 15 secret <choose-a-password>
crypto key generate rsa modulus 2048
ip ssh version 2
interface FastEthernet0/0
 description MANAGEMENT_DHCP
 ip address dhcp
 no shutdown
 exit
line vty 0 4

 transport input ssh
 exit
